+/**

+ * install the same opaque-access, Clerk liveness and phase-bound readers, and
+ * return only after successful lease recheck and commit. pool and clerk are
+ * request-owned; nothing selects a reader through HTTP, env or query values.
+ * Passing a callable is not a deployment approval or ACL verification.
+ */
+export async function handleCabinetAccess(request,{withCabinetReaders,pool,clerk}={}){
+  const routeFault=cabinetAccessRequestFault(request);
+  if(routeFault)return routeFault;
+  if(request.signal.aborted||typeof withCabinetReaders!=="function"||!pool||typeof clerk?.liveness!=="function")
+    return unavailable();
+  const accessToken=request.headers.get("authorization")?.match(token)?.[1];
+  if(!accessToken||accessToken!==accessToken.toLowerCase())return reject("SESSION_REVOKED",401);
+  let called=0,settled=false,decision,finished=false;
+  try{
+    const returned=await withCabinetReaders(Object.freeze({
+      accessToken,signal:request.signal,pool,clerk,
+    }),async readers=>{
+      if(finished||++called!==1||request.signal.aborted)throw Error("CABINET_READ_UNAVAILABLE");
+      decision=await readAuthorizedCabinetEntry(accessToken,request.signal,readers);
+      settled=true;
+      return decision;
+    });
+    if(request.signal.aborted||called!==1||!settled||returned!==decision)return unavailable();
+    if(decision===null)return reject("SESSION_REVOKED",401);
+    return json(200,{data:decision});
+  }catch{return unavailable();}
+  finally{finished=true;}
+}
