+const unavailable=()=>new ApiFault("SERVICE_UNAVAILABLE",503);
+const validRole=role=>typeof role==="string"&&/^[a-z_][a-z0-9_]{0,62}$/.test(role);
+function own(object,field){
+  const value=object&&Object.getOwnPropertyDescriptor(object,field);
+  return value&&Object.hasOwn(value,"value")?value.value:undefined;
+}

+/**
+ * Optional trusted assembly; the production/default Worker does not call it.
+ * The two policy checks are explicit confirmation from reviewed server
+ * composition, not evidence supplied by this module. The role check below
+ * catches an accidentally shared DSN username; it does not attest actual DB
+ * privileges, RLS completeness, upstream identity or Hyperdrive configuration.
+ * Parent must separately verify dedicated role grants, bound runtime identity,
+ * current-attempt RLS parity, no caching, TLS and bounded driver disposal.
+ */
+export function createOptInCabinetReaderAssembly(options){
+  const StaffClient=own(options,"StaffClient");
+  const dedicatedPostgres=own(options,"dedicatedPostgres");
+  const expectedStaffRole=own(options,"expectedStaffRole");
+  const timeoutMs=own(options,"timeoutMs");
+  if(own(options,"currentAttemptPolicyVerified")!==true||own(options,"staffCapabilityPolicyVerified")!==true||
+    typeof StaffClient!=="function"||typeof dedicatedPostgres!=="function"||!validRole(expectedStaffRole)||
+    !Number.isSafeInteger(timeoutMs)||timeoutMs<100||timeoutMs>5000)return undefined;
+
+  return async function withCabinetReaders(scope,use){
+    if(!(scope?.signal instanceof AbortSignal)||scope.signal.aborted||
+      !validRole(scope.studentConnectionRole)||scope.studentConnectionRole===expectedStaffRole)
+      throw unavailable();
+    // No request, app proof, digest or identity tuple is passed to the supplier.
+    let options;
+    try{options=dedicatedPostgres();}catch{throw unavailable();}
+    const connectionString=own(options,"connectionString");
+    if(!options||![Object.prototype,null].includes(Object.getPrototypeOf(options))||Reflect.ownKeys(options).length!==1||
+      !isHyperdriveConnectionString(connectionString))throw unavailable();
+    try{
+      if(decodeURIComponent(new URL(connectionString).username)!==expectedStaffRole)throw unavailable();
+    }catch{throw unavailable();}
+
+    let staffResource,staffSignal,selected=false;
+    const run=createCabinetReadersTransaction({
+      currentAttemptPolicyVerified:true,
+      async readStaffCapability(actor,signal){
+        if(selected||signal!==scope.signal||signal.aborted)throw unavailable();
+        selected=true;
+        staffSignal=AbortSignal.any([signal,AbortSignal.timeout(timeoutMs)]);
+        staffResource=createCloudflarePostgres(StaffClient,{connectionString},{signal:staffSignal});
+        if(staffResource.pool===scope.pool)throw unavailable();
+        let staffClient,usable=false;
+        try{
+          staffClient=await staffResource.pool.connect();
+          if(staffSignal.aborted)throw unavailable();
+          const verdict=await createCabinetStaffCapabilityReader(staffClient)(actor,staffSignal);
+          if(staffSignal.aborted)throw unavailable();
+          usable=true;
+          return verdict;
+        }finally{
+          staffClient?.release(!usable);
+        }
+      }
+    });
+    let result;
+    try{result=await run(scope,use);}
+    finally{if(staffResource)await staffResource.close();}
+    if(scope.signal.aborted||staffSignal?.aborted)throw unavailable();
+    return result;
+  };
+}
