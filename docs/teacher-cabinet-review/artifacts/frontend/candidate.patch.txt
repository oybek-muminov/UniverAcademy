+import {useEffect,useRef,useState} from 'react';
+import type {ReactElement} from 'react';

+type Input = Readonly<{personId:string;signal:AbortSignal}>;
+type PreparedTeacher = Readonly<{personId:string;view:ReactElement}>;
+
+// Review-only port. It has no default implementation or production configuration.
+// personId is an expected local identity, never URL/query/header authorization.
+// resolveAccess must derive identity and current role from the trusted session and
+// include authoritative server-side unfinished-attempt continuity across launches.
+// prepareTeacher must prove the same person in the destination independently and
+// return an unmounted view. Its preflight signal must not control that view's lifetime.
+// It must not navigate or exchange/clear Student proofs.
+// The Teacher server must still authorize every data read and mutation.
+export type TeacherCabinetReviewPort = Readonly<{
+  resolveAccess(input:Input):Promise<unknown>;
+  prepareTeacher(input:Input):Promise<PreparedTeacher|null>;
+}>;
+
+type Selection =
+  | {kind:'pending'}
+  | {kind:'student'}
+  | {kind:'stale'}
+  | {kind:'teacher';view:ReactElement};
+
+function permitsTeacher(value:unknown,personId:string):boolean {
+  if(value===null||typeof value!=='object'||Object.getPrototypeOf(value)!==Object.prototype)return false;
+  const keys=Reflect.ownKeys(value);
+  if(keys.length!==2||!keys.includes('person_id')||!keys.includes('default_cabinet'))return false;
+  return Object.getOwnPropertyDescriptor(value,'person_id')?.value===personId&&
+    Object.getOwnPropertyDescriptor(value,'default_cabinet')?.value==='teacher';
+}

+const DECISION_BUDGET_MS=650;

+export function TeacherCabinetEntryReview(props:{
+  personId:string;

+  port:TeacherCabinetReviewPort;
+  isCurrent():boolean;
+  student:ReactElement;
+}){

+  }));

+  const closed=useRef(false);
+  const stillCurrent=props.identityKey===entry.identityKey&&props.personId===entry.personId&&
+    props.port===entry.port&&entry.isCurrent();
+
+  useEffect(()=>{
+    if(!stillCurrent){closed.current=true;setSelection({kind:'stale'});return}
+    if(closed.current)return;
+    const abort=new AbortController();
+    const deadline=performance.now()+DECISION_BUDGET_MS;
+    const commit=(next:Selection)=>{
+      if(abort.signal.aborted||closed.current)return;
+      const result:Selection=!entry.isCurrent()?{kind:'stale'}:
+        next.kind==='teacher'&&performance.now()>=deadline?{kind:'student'}:next;
+      closed.current=true;
+      if(result.kind!=='teacher')abort.abort();
+      setSelection(current=>current.kind==='pending'?result:current);
+    };
+    const timer=setTimeout(()=>commit({kind:'student'}),DECISION_BUDGET_MS);
+    void (async()=>{
+      try{
+        const input={personId:entry.personId,signal:abort.signal};
+        const access=await entry.port.resolveAccess(input);
+        if(abort.signal.aborted)return;
+        if(!entry.isCurrent()||!permitsTeacher(access,entry.personId)){commit({kind:'student'});return}
+        const teacher=await entry.port.prepareTeacher(input);
+        if(teacher?.personId===entry.personId)commit({kind:'teacher',view:teacher.view});
+        else commit({kind:'student'});
+      }catch{commit({kind:'student'})}
+      finally{clearTimeout(timer)}
+    })();
+    return()=>{clearTimeout(timer);abort.abort()};
+  },[entry,stillCurrent]);
+
+  if(!stillCurrent||selection.kind==='stale')return <main className="shell"><section className="card">
+    <p role="status">Telegram botida /start orqali yangidan oching.</p>
+  </section></main>;
+  if(selection.kind==='pending')return <main className="shell"><section className="card">
+    <p role="status">Kabinet tekshirilmoqda…</p>
+  </section></main>;
+  if(selection.kind==='teacher')return selection.view;
+  return props.student;
