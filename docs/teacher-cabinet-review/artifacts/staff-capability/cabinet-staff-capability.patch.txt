+
+const uuid=/^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/;
+const capability=`
+SELECT EXISTS (
+  SELECT 1
+  FROM ua_private.clerk_identities AS i
+  JOIN ua_private.people AS p ON p.id=i.person_id
+  JOIN ua_private.staff_access AS a ON a.person_id=p.id
+  WHERE i.person_id=$1::uuid AND i.issuer=$2 AND i.subject=$3
+    AND i.enabled IS TRUE AND p.enabled IS TRUE
+    AND 'teacher'=ANY(p.roles) AND a.approved IS TRUE
+    AND EXISTS (
+      SELECT 1 FROM ua_private.teacher_subject_approvals AS t
+      WHERE t.person_id=p.id AND t.approved IS TRUE
+        AND t.revoked_at_ms IS NULL AND t.subject_id<>''
+    )
+) AS can_open_teacher`;
+
+const unavailable=():never=>fail("SERVICE_UNAVAILABLE",503);
+
+/**
+ * Unmounted, Worker-only adapter. The caller injects a separate request-owned
+ * capability connection whose role has only reviewed read privileges, sends no
+ * app proof/digest, and owns cancellation, timeout and disposal. Never use
+ * the Student transaction client here. One unlocked statement on the other
+ * connection is not atomic with the Student lock/snapshot and must not request
+ * a person lock. The caller rechecks its own Clerk lease before accepting it.
+ */
+export function createCabinetStaffCapabilityReader(
+  client:Pick<PgClient,"query">,
+):CabinetStaffCapabilityReader{
+  if(typeof client?.query!=="function")return unavailable();
+  return async(actor,signal)=>{
+    if(signal.aborted||!actor||typeof actor.personId!=="string"||!uuid.test(actor.personId)||
+      typeof actor.issuer!=="string"||actor.issuer.length>512||!/^https:\/\/[^/\s]+$/.test(actor.issuer)||
+      typeof actor.subject!=="string"||actor.subject.length>256||!/^user_[A-Za-z0-9]+$/.test(actor.subject))
+      return unavailable();
+    let rows:Awaited<ReturnType<PgClient["query"]>>["rows"];
+    try{
+      ({rows}=await client.query({name:"ua_cabinet_staff_capability_v1",text:capability,
+        values:[actor.personId,actor.issuer,actor.subject]}));
+    }catch{return unavailable();}
+    if(signal.aborted||!Array.isArray(rows)||rows.length!==1)return unavailable();
+    const row=rows[0];
+    if(!row||Reflect.ownKeys(row).length!==1||
+      typeof Object.getOwnPropertyDescriptor(row,"can_open_teacher")?.value!=="boolean")return unavailable();
+    return row.can_open_teacher as boolean;
+  };
+}
